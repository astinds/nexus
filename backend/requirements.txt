# NEXUS backend runtime dependencies — the canonical list.
#
# This file lives inside backend/ because render.yaml sets `rootDir: backend`
# and runs `pip install -r requirements.txt` relative to it. Keep it here, not
# only at the repo root, or the Render build fails with "no such file".
#
# Target: Render Free Web Service (Linux x86_64/glibc), where every entry below
# resolves to a prebuilt wheel. On Android/Termux use requirements-dev.txt
# instead: PyMuPDF and psycopg[binary] have no bionic wheels there.

fastapi>=0.115,<1.0
uvicorn>=0.30,<1.0
pydantic>=2.7,<3.0
pydantic-settings>=2.2,<3.0
python-multipart>=0.0.9
httpx>=0.27,<1.0

# Database. [binary] ships the libpq C extension as a wheel, so Render needs no
# system libpq-dev and no compiler at build time.
psycopg[binary]>=3.2,<4.0
psycopg-pool>=3.2,<4.0

# Extractors
# Pinned below 2.0 on purpose. pymupdf 2.x drags pymupdf4llm, pymupdf-layout,
# onnxruntime, flatbuffers and protobuf into the build (~100 MB of wheels) for a
# feature this app never imports, on a 512 MB Render Free instance. The
# extractor only uses `pymupdf.open` + `page.get_text`, which 1.x serves.
pymupdf>=1.24,<2.0       # primary PDF engine: 1-based pages + scanned detection
pypdf>=5.0               # pure-Python PDF fallback (kept for parity and safety)
python-docx>=1.1
beautifulsoup4>=4.12
lxml>=5.2
trafilatura>=1.9         # web article extraction; falls back to BeautifulSoup
